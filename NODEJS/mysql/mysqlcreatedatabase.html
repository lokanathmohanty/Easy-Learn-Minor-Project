Node.js - MySQL Create Database

Overview

Multiple databases (schemas) can be created on a MySQL server. This example demonstrates how to create a database using Node.js with the MySQL driver.

MySQL Query Syntax

CREATE DATABASE IF NOT EXISTS database_name;

This creates the database only if it does not already exist.

Example: Create Database with Node.js

var mysql = require('mysql');

var con = mysql.createConnection({
  host: "localhost",
  user: "root",
  password: "mypassword"
});

con.connect(function (err) {
  if (err) throw err;
  console.log("Connected!");
  
  // Create database named 'mydb'
  con.query("CREATE DATABASE IF NOT EXISTS mydb;", function (err, result) {
    if (err) throw err;
    console.log("Created database successfully");
  });
  
  // Show all databases
  con.query("SHOW DATABASES;", function (err, result) {
    if (err) throw err;
    console.log("Showing databases\n");
    for (var i = 0; i < result.length; i++) {
      console.log(JSON.stringify(result[i]));
    }
  });
  
  con.end();
});
