Node.js - MongoDB Join

Overview

MongoDB, being a NoSQL database, does not support direct JOIN operations as in relational databases. However, similar functionality is achievable in Node.js using the aggregate() method with the $lookup stage.

$lookup Stage Syntax

{
  $lookup: {
    from: <collection to join>,
    localField: <field in current collection>,
    foreignField: <field in foreign collection>,
    as: <output array field>
  }
}

This performs a left outer join on the specified collections and adds matching documents as an array in the output field.

Example: Joining 'orders' and 'inventory' Collections

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);

  try {
    await client.connect();
    await joinDocs(client, "mydb", "orders", "inventory");
  } finally {
    await client.close();
  }
}

async function joinDocs(client, dbname, col1, col2) {
  const result = await client.db(dbname).collection(col1).aggregate([
    {
      $lookup: {
        from: col2,
        localField: "prodId",
        foreignField: "prodId",
        as: "orderdetails"
      }
    }
  ]).toArray();
  
  result.forEach(doc => console.log(JSON.stringify(doc)));
}

main().catch(console.error);
