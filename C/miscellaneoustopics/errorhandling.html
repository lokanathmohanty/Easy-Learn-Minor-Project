Error Handling in C

C does not provide direct support for error handling with keywords like exceptions. Instead, programmers use functions and global variables such as errno for handling errors.

The errno Variable

Most C or Unix function calls return -1 or NULL on error and set the global variable errno to indicate the error type. The error codes are defined in the <errno.h> header file.

Initializing errno to 0 is a good practice. Zero means no error.

Common errno values

errno value Error
1Operation not permitted
2No such file or directory
3No such process
4Interrupted system call
5I/O error
6No such device or address
7The argument list is too long
8Exec format error
9Bad file number
10No child processes
11Try again
12Out of memory
13Permission denied

Example

Opening a non-existing file sets errno:

#include <stdio.h>
#include <errno.h>

int main() {
   FILE* fp;

   fp = fopen("nosuchfile.txt", "r");
   printf("Value of errno: %d\n", errno);

   return 0;
}

The perror() Function

Displays the string passed to it, followed by a colon, a space, and the textual representation of the current errno value.

void perror(const char *str);

Example

#include <stdio.h>
#include <errno.h>

int main(){
   FILE* fp;

   fp = fopen("nosuchfile.txt", "r");

   printf("Value of errno: %d\n", errno);
   perror("Error message:");

   return 0;
}

The strerror() Function

Returns a pointer to the textual representation of a given errno value.

char *strerror(int errnum);

Example

#include <stdio.h>
#include <errno.h>

int main() {
   FILE* fp;

   fp = fopen("nosuchfile.txt", "r");

   printf("Value of errno: %d\n", errno);
   printf("The error message is : %s\n", strerror(errno));

   return 0;
}

The ferror() Function

Checks if an error occurred during a file operation.

int ferror(FILE *stream);

Example

#include <stdio.h>

int main(){
   FILE *fp;
   fp = fopen("test.txt","w");
   char ch = fgetc(fp);  // Trying to read from writable file
   if(ferror(fp)){
      printf("File is opened in writing mode! You cannot read data from it!");
   }
   fclose(fp);
   return 0;
}

The clearerr() Function

Clears end-of-file and error indicators for a file stream.

void clearerr(FILE *stream);

Example

#include <stdio.h>

int main(){
   FILE *fp;
   fp = fopen("test.txt","w");

   char ch = fgetc(fp);  // Trying to read from writable file

   if(ferror(fp)){
      printf("File is opened in writing mode! You cannot read data from it!\n");
   }

   clearerr(fp);

   if(ferror(fp)){
      printf("Error again in reading from file!");
   }
   fclose(fp);
   return 0;
}

Divide by Zero Errors

Check the divisor before division to avoid runtime errors.

Example 1

#include <stdio.h>
#include <stdlib.h>

int main() {
   int dividend = 20;
   int divisor = 0;
   int quotient;

   if(divisor == 0){
      fprintf(stderr, "Division by zero! Exiting...\n");
      exit(-1);
   }
   quotient = dividend / divisor;
   fprintf(stderr, "Value of quotient : %d\n", quotient );
   exit(0);
}

Example 2

Use EXIT_SUCCESS and EXIT_FAILURE macros for program exit status.

#include <stdio.h>
#include <stdlib.h>

int main() {
   int dividend = 20;
   int divisor = 5;
   int quotient;

   if(divisor == 0) {
      fprintf(stderr, "Division by zero! Exiting...\n");
      exit(EXIT_FAILURE);
   }

   quotient = dividend / divisor;
   fprintf(stderr, "Value of quotient: %d\n", quotient );

   exit(EXIT_SUCCESS);
}
