Node.js - MySQL Insert Into

Overview

In a MySQL table, records (rows) store data with columns (attributes) as defined by the table structure. The INSERT INTO statement is used to add new rows.

MySQL Insert Syntax

INSERT INTO table_name (field1, field2, ...)
VALUES (value1, value2, ...);

If the values match the table structure's order and count, the field list can be omitted. Auto-increment fields are filled automatically.

Insert Example in MySQL CLI

INSERT INTO employee VALUES (1, "Ravi", 25, 25000);

SELECT * FROM employee;

Output:

+----+-------+-----+--------+
| id | name  | age | salary |
+----+-------+-----+--------+
|  1 | Ravi  |  25 |  25000 |
+----+-------+-----+--------+

Insert Using Node.js

Load database in connection:

var mysql = require('mysql');

var con = mysql.createConnection({
  host: "localhost",
  user: "root",
  password: "mypassword",
  database: "mydb"
});

Simple Insert

var qry = `INSERT INTO employee (name, age, salary)
VALUES ("Ravi", 25, 25000);`;

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, function(err, result) {
    if (err) throw err;
    console.log("New Record inserted successfully");
  });
  con.end();
});

Prepared Statement Insert

Use placeholders ? and pass values as array for safer queries:

var data = ["Ravi", 25, 25000];
var qry = `INSERT INTO employee (name, age, salary) VALUES (?, ?, ?);`;

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, data, function(err, result) {
    if (err) throw err;
    console.log("New Record inserted successfully");
  });
  con.end();
});

Insert Multiple Records

var data = [
  ["Ravi", 25, 25000],
  ["Anil", 26, 30000],
  ["Meena", 26, 27000]
];

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  for (var i = 0; i < data.length; i++) {
    con.query(qry, data[i], function(err, result) {
      if (err) throw err;
      console.log("New Record inserted successfully");
    });
  }
  con.end();
});
