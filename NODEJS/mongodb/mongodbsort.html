Node.js - MongoDB Sort

Overview

In MongoDB, results of a query can be sorted in ascending or descending order on specified document fields using the sort() method of the Collection object in the Node.js driver.

Syntax

- Ascending order: sort({field: 1})
- Descending order: sort({field: -1})

Example: Ascending Sort

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);

  try {
    await client.connect();
    await sortDocs(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function sortDocs(client, dbname, colname) {
  const sortOrder = { price: 1 }; // ascending
  const cursor = client.db(dbname).collection(colname)
                          .find({})
                          .sort(sortOrder);
  const results = await cursor.toArray();
  results.forEach(doc => console.log(doc));
}

main().catch(console.error);

                          .find({})
