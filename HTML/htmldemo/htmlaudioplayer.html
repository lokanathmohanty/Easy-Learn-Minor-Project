HTML Audio Player with Visualizer

Introduction

HTML provides native support for audio and video without requiring Flash. Using the Web Audio API, we can create an audio player with a visualizer that reacts to the frequency data of the playing sound. The visualizer will display bars or waves as the music plays. In this example, users can drag and drop local MP3 files into the player container.

Code Files Required

We need two files to build a custom HTML audio player with visualizer:

  • An HTML file to define the structure.
  • A JavaScript file to handle audio processing and visualization.

1. HTML File

<!DOCTYPE html>
<html>
<body>
   <div id="container">
      <div id="canvas-container">
         <canvas width="600" height="300" id="canvas"></canvas>
         <canvas width="600" height="300" id="canvas-copy"></canvas>
      </div>
      <div id="instructions">
         <a href="https://www.tutorialspoint.com/index.htm">Tutorials Point</a>
         <h2>Drag Your Local MP3 Files</h2>
      </div>
   </div>
   <script src="js.html"></script>
</body>
</html>

2. JavaScript File

(function() {
   var requestAnimationFrame = 
      window.requestAnimationFrame || 
      window.mozRequestAnimationFrame || 
      window.webkitRequestAnimationFrame || 
      window.msRequestAnimationFrame;
   window.requestAnimationFrame = requestAnimationFrame;
})();

window.onload = function() {
   var element = document.getElementById('container');
   dropAndLoad(element, init, "ArrayBuffer");
}

function dropAndLoad(dropElement, callback, readFormat) {
   dropElement.addEventListener('dragover', function(e) {
      e.preventDefault();
   }, false);

   dropElement.addEventListener('drop', function(e) {
      e.preventDefault();
      loadFile(e.dataTransfer.files[0]);
   }, false);

   function loadFile(file) {
      var reader = new FileReader();
      reader.onload = function(e) {
         callback(e.target.result);
      }
      reader['readAs' + readFormat](file);
   }
}

function init(arrayBuffer) {
   document.getElementById('instructions').innerHTML = 'Audio Loading';
   var audioCtx = new AudioContext();
   var analyser = audioCtx.createAnalyser();
   audioCtx.decodeAudioData(arrayBuffer, function(buffer) {
      var source = audioCtx.createBufferSource();
      source.buffer = buffer;
      source.connect(analyser);
      analyser.connect(audioCtx.destination);
      source.start(0);

      var viz = new simpleViz();
      new visualizer(viz.update, analyser);
      document.getElementById('instructions').innerHTML = '';
   });
}

function visualizer(visualization, analyser) {
   function loop() {
      var byteFreq = new Uint8Array(analyser.frequencyBinCount);
      analyser.getByteFrequencyData(byteFreq);
      visualization(byteFreq);
      requestAnimationFrame(loop);
   }
   requestAnimationFrame(loop);
}

function simpleViz() {
   var canvas = document.getElementById('canvas');
   var ctx = canvas.getContext("2d");
   var copyCtx = document.getElementById('canvas-copy').getContext("2d");
   var barWidth = 10, barGap = 4;
   var bars = Math.floor(canvas.width / (barWidth + barGap));

   this.update = function(byteFreq) {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      var step = Math.floor(byteFreq.length / bars);
      for (var i = 0; i < bars; i++) {
         var barHeight = byteFreq[i * step];
         ctx.fillRect(i * (barWidth + barGap), canvas.height - barHeight, barWidth, barHeight);
         copyCtx.clearRect(0, 0, canvas.width, canvas.height);
         copyCtx.drawImage(canvas, 0, 0);
      }
   }
}

Complete Audio Player

When combined, the above HTML and JavaScript files create an audio player with a bar visualizer. Dragging a local MP3 file into the container will play it with a synchronized visual effect.

HTML Audio Player with Playlist

We can extend the functionality by allowing multiple MP3 uploads to form a playlist:

<!DOCTYPE html>
<html>
<body>
   <audio controls id="player" autoplay></audio>
   <input type="file" id="fileInput" multiple>

   <script>
      var fileInput = document.getElementById("fileInput"),
          player = document.getElementById("player"),
          index = 0, files;

      function playNext(n) {
         var url = URL.createObjectURL(files[n]);
         player.src = url;
         player.play();
      }

      fileInput.addEventListener('change', function() {
         files = fileInput.files;
         if (files.length) playNext(index);
      });

      player.addEventListener("ended", function() {
         index++;
         if (index < files.length) playNext(index);
      });
   </script>
</body>
</html>

This example allows users to upload multiple MP3s, which then play automatically as a playlist.
