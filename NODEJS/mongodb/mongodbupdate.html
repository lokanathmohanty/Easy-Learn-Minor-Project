Node.js - MongoDB Update

Overview

MongoDB update operations modify values in documents within a collection. The Node.js driver provides updateOne() for single document updates and updateMany() for multiple document updates.

updateOne() Syntax

collection.updateOne(filter, update);

The filter finds the document to update, and update specifies the changes.

Example: Update Single Document

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);

  try {
    await client.connect();
    await updateOneProduct(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function updateOneProduct(client, dbname, colname) {
  const filter = { ProductID: 3 };
  const update = { $set: { Name: "Router", price: 2750 } };
  const result = await client.db(dbname).collection(colname).updateOne(filter, update);
  console.log("Document updated");
}

main().catch(console.error);

Verification in Mongo Shell:

> use mydb
> db.products.find({ ProductID: 3 })
{
  _id: ObjectId("..."),
  ProductID: 3,
  Name: "Router",
  price: 2750
}

updateMany() Example

Increments price by 125 for products with names ending in 'er'.

async function updateManyProducts(client, dbname, colname) {
  const filter = { Name: /er$/ };
  const update = { $inc: { price: 125 } };
  const result = await client.db(dbname).collection(colname).updateMany(filter, update);
  console.log("Documents updated");
}

Output:

Documents updated
