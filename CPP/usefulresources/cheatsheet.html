C++ Cheat Sheet: Core Syntax & Modern Features 🚀

I. Compilation & Basic Structure

// 1. Compile & Run (Linux/macOS)
g++ file.cpp -o program
./program

// 2. Basic Structure
#include <iostream> // Header for input/output

int main() {
    // Standard output
    std::cout << "Hello" << std::endl; 
    
    // Standard input
    int x;
    std::cin >> x; 
    
    return 0; // Success
}

II. Control Flow

// 1. If/Else
if (condition) {
    // code
} else if (other_condition) {
    // code
} else {
    // code
}

// 2. For Loop
for (int i = 0; i < 10; ++i) {
    // 10 iterations
}

// 3. Range-based For (Modern C++)
std::vector<int> v = {1, 2, 3};
for (int val : v) { 
    // val is a COPY
}
for (int& val : v) {
    // val is a REFERENCE (can modify original)
}

// 4. While Loop
while (condition) {
    // code
}

III. Memory, Pointers, and References

int a = 10;

// 1. Pointer (stores address)
int* ptr = &a; // & is the "address of" operator
std::cout << *ptr; // * is the "dereference" operator (access value)

// 2. Reference (an alias)
int& ref = a; // Must be initialized
ref = 20;    // Changes 'a' to 20

// 3. Dynamic Memory (Heap)
int* dynamic_int = new int(42);
delete dynamic_int;
dynamic_int = nullptr; // Always set to nullptr after delete

IV. Modern C++ Features (C++11+)

Smart Pointers (RAII)

#include <memory>
// Unique ownership; cannot be copied
std::unique_ptr<int> p1(new int(5)); 
// Shared ownership; reference counted
std::shared_ptr<int> p2 = std::make_shared<int>(10);

Lambdas (Anonymous Functions)

int x = 5;
int y = 10;

// [capture] (params) -> return_type { body }
auto add = [](int a, int b) { return a + b; };
std::cout << add(1, 2); // Output: 3

// Capture 'x' by reference (&), 'y' by value (=)
auto use_vars = [=, &x]() { 
    x = x + y; // Modifies external x
    // y = 20; // ERROR: y is const (captured by value)
};
use_vars(); // x is now 15

Standard Template Library (STL)

#include <vector>
#include <string>
#include <map>
#include <algorithm>

// Vector (Dynamic Array)
std::vector<int> nums = {5, 2, 8};
nums.push_back(1); // Add to end
std::sort(nums.begin(), nums.end()); // Use <algorithm>

// Map (Key-Value/Dictionary)
std::map<std::string, int> ages;
ages["Alice"] = 30;
std::cout << ages["Alice"];

// Unordered Containers (Hash Tables - fast O(1) average lookup)
std::unordered_set<int> s = {1, 2, 2}; // Stores unique elements
std::unordered_multiset<int> ms = {1, 2, 2}; // Allows duplicates
