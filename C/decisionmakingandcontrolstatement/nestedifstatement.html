Nested If Statements in C

In C programming, nested if statements refer to using one if or else-if statement inside another. This is useful for building multi-level decision-making logic.

Syntax

if (expr1){
   if (expr2){
      // executes if expr1 and expr2 are true
   }
   else{
      // executes if expr1 is true but expr2 is false
   }        
}

Nested if statements can appear inside either the if block or else block, or both.

Example 1: Multi-range number classification

#include <stdio.h>

int main() {
   int a = 274;
   printf("Value of a is : %d\n", a);

   if (a < 100){
      printf("Value of a is less than 100\n");
   }
   
   if (a >= 100 && a < 200){
      printf("Value of a is between 100 and 200\n");
   }
      
   if (a >= 200){
      printf("Value of a is more than 200\n");
   }
}

Output:
Value of a is : 274
Value of a is more than 200

Example 2: Same logic using nested if

#include <stdio.h>

int main() {
   int a = 120;
   printf("Value of a is : %d\n", a );
   if(a >= 100){
      if(a < 200){
         printf("Value of a is between 100 and 200\n");
      }
      else{
         printf("Value of a is more than 200\n");
      }
   }
   else{
      printf("Value of a is less than 100\n");
   }
   return 0;
}

Output:
Value of a is : 120
Value of a is between 100 and 200

Example 3: Divisibility Check

#include <stdio.h>

int main(){
   int a = 15;
   printf("a: %d\n", a);

   if (a % 2 == 0) {
      if (a % 3 == 0){
         printf("Divisible by 2 and 3");
      }
      else {
         printf("Divisible by 2 but not 3");
      }
   }
   else {
      if (a % 3 == 0){
         printf("Divisible by 3 but not 2");
      }
      else{
         printf("Not divisible by 2, not divisible by 3");
      }
   }
   return 0;
}

Output:
a: 15
Divisible by 3 but not 2

Example 4: Leap Year Check

#include <stdio.h>

int main(){
   int year = 1900;
   printf("year: %d\n", year);
   if (year % 4 == 0){
      if (year % 100 == 0){
         if(year % 400 == 0){
            printf("%d is a Leap Year\n", year);
         }
         else{
            printf("%d is not a Leap Year\n", year);
         }
      }
      else{
         printf("%d is a Leap Year\n", year);
      }
   }
   else{
      printf("%d is not a Leap Year\n", year);
   }
   return 0;
}

Output:
year: 1900
1900 is not a Leap Year

Alternative: Compound Boolean Expression

if (year % 4 == 0 && (year % 400 == 0 || year % 100 != 0)) {
   printf("%d is a leap year", year);
} else {
   printf("%d is not a leap year", year);
}

Nested if statements help write structured, easy-to-read multi-level decision logic for complex algorithms.
