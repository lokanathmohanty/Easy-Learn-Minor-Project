JavaScript - IndexedDB

What is IndexedDB?

IndexedDB is a low-level JavaScript API for storing and retrieving large amounts of structured data, including files and blobs. It allows web applications to locally store, query, and modify data on the user's device. IndexedDB is useful for applications handling large datasets, working offline, and providing responsive user experiences with local caching.

Key Concepts and Features

  • Asynchronous API − Uses event-driven programming to avoid blocking the main thread.
  • Database − A container for object stores, used for organizing and retrieving data.
  • Object Store − Similar to a table in relational databases, storing collections of JavaScript objects and supporting CRUD operations.
  • Index − Provides faster querying and sorting based on specific fields.
  • Transactions − Ensures consistency and integrity by grouping multiple operations into a single unit that either succeeds or fails together.

Why use IndexedDB?

IndexedDB is ideal for client-side storage because:

  • It prevents main-thread blocking with its asynchronous nature.
  • It supports transactions to maintain data integrity.
  • It handles large volumes of structured data locally.
  • It improves offline functionality by reducing server communication.
  • It supports flexible key structures, indexing, and schema upgrades.

CRUD Operations

Create / Insert

const request = objectStore.add(data);
request.onsuccess = () => {
  // Handle success
};
request.onerror = () => {
  // Handle error
};

Read (Retrieve)

Using get()

Retrieves a single record based on a known key.

Using openCursor()

const request = objectStore.openCursor();
request.onsuccess = (event) => {
  const cursor = event.target.result;
  if (cursor) {
    const data = cursor.value;
    // Process data
    cursor.continue();
  }
};
request.onerror = () => {
  // Handle error
};

Update

The put() method updates a record if it exists, or adds it if it doesn’t.

const request = objectStore.put(data);

Delete

The delete() method removes a record by key.

const request = objectStore.delete(key);

Implementation Example

The following example demonstrates IndexedDB CRUD operations with a table interface. It allows creating, updating, and deleting records. A form is displayed when creating new data, and updates are done via prompts.

Example: IndexedDB CRUD Operations

<!DOCTYPE html>
<html>
<head>
  <title>IndexedDB CRUD</title>
</head>
<body>
  <h2>IndexedDB CRUD Operations</h2>
  <button onclick="showCreateForm()">Create Data</button>
  <div id="createForm" style="display:none;">
    <h3>Create Data</h3>
    <label for="name">Name:</label>
    <input type="text" id="name"><br><br>
    <label for="email">Email:</label>
    <input type="email" id="email"><br>
    <button onclick="createData()">Save</button>
    <button onclick="cancelCreate()">Cancel</button>
  </div>
  <table id="data-table">
    <thead>
      <tr>
        <th>ID</th>
        <th>Name</th>
        <th>Email</th>
        <th>Actions</th>
      </tr>
    </thead>
    <tbody></tbody>
  </table>

  <script>
    const dbName = "myDatabase";
    let db;
    const request = indexedDB.open(dbName, 1);

    request.onupgradeneeded = (event) => {
      db = event.target.result;
      const store = db.createObjectStore("myObjectStore", { keyPath: "id", autoIncrement: true });
      store.createIndex("name", "name", { unique: false });
      store.createIndex("email", "email", { unique: true });
    };

    request.onsuccess = (event) => {
      db = event.target.result;
      showData();
    };

    function showData() {
      const transaction = db.transaction(["myObjectStore"], "readonly");
      const store = transaction.objectStore("myObjectStore");
      const tableBody = document.querySelector("#data-table tbody");
      tableBody.innerHTML = "";
      const request = store.openCursor();
      request.onsuccess = (event) => {
        const cursor = event.target.result;
        if (cursor) {
          const row = tableBody.insertRow();
          row.insertCell(0).textContent = cursor.value.id;
          row.insertCell(1).textContent = cursor.value.name;
          row.insertCell(2).textContent = cursor.value.email;
          const actionsCell = row.insertCell(3);
          actionsCell.innerHTML = `
            <button onclick="showUpdateForm(${cursor.value.id})">Update</button>
            <button onclick="deleteData(${cursor.value.id})">Delete</button>
          `;
          cursor.continue();
        }
      };
    }

    function createData() {
      const transaction = db.transaction(["myObjectStore"], "readwrite");
      const store = transaction.objectStore("myObjectStore");
      const name = document.getElementById("name").value;
      const email = document.getElementById("email").value;
      const newData = { name, email };
      const request = store.add(newData);
      request.onsuccess = () => {
        showData();
        cancelCreate();
        alert("Data added successfully");
      };
    }

    function showCreateForm() {
      document.getElementById("createForm").style.display = "block";
    }

    function cancelCreate() {
      document.getElementById("createForm").style.display = "none";
      document.getElementById("name").value = "";
      document.getElementById("email").value = "";
    }

    function showUpdateForm(id) {
      const transaction = db.transaction(["myObjectStore"], "readwrite");
      const store = transaction.objectStore("myObjectStore");
      const request = store.get(id);
      request.onsuccess = (event) => {
        const data = event.target.result;
        if (data) {
          const name = prompt("Update Name:", data.name);
          const email = prompt("Update Email:", data.email);
          if (name !== null && email !== null) {
            data.name = name;
            data.email = email;
            const updateRequest = store.put(data);
            updateRequest.onsuccess = () => {
              showData();
              alert("Updated data for " + data.name);
            };
          }
        }
      };
    }

    function deleteData(id) {
      const transaction = db.transaction(["myObjectStore"], "readwrite");
      const store = transaction.objectStore("myObjectStore");
      const request = store.delete(id);
      request.onsuccess = () => {
        showData();
        alert("Deleted data for id " + id);
      };
    }
  </script>
</body>
</html>

Try running the above program. You’ll see an interface to create, update, and delete data. The data is displayed in tabular format after each operation.
