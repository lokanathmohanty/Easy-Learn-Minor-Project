ReactJS - Custom Hooks

Hooks are integral part of React function components. While React provides several built-in hooks, sometimes developers need specialized logic. React allows creating custom hooks by extracting and reusing logic across components.

Create a Custom Hook

We’ll build an infinite scroll feature, first inside a component, then extract it as a reusable useInfiniteScroll custom hook.

Step 1: Implement Infinite Scroll in a Component

Start with a TodoList component that generates dummy items. As the user scrolls to the bottom, more items are appended. 

import React, { useState, useEffect, useMemo } from "react";

function TodoList() {
   const [count, setCount] = useState(100);
   let data = [];

   useMemo(() => {
      for (let i = 1; i <= count; i++) {
         data.push({ id: i, todo: "Todo Item " + i });
      }
   }, [count]);

   useEffect(() => {
      function handleScroll() {
         const isBottom =
            window.innerHeight + document.documentElement.scrollTop
            === document.documentElement.offsetHeight;

         if (isBottom) setCount(count + 100);
      }
      window.addEventListener("scroll", handleScroll);
      return () => window.removeEventListener("scroll", handleScroll);
   }, []);

   return (
      <div>
         <p>List of Todo list</p>
         <ul>
            {data.map(item => <li key={item.id}>{item.todo}</li>)}
         </ul>
      </div>
   );
}
export default TodoList;

Step 2: Extract Infinite Scroll into a Custom Hook

Next, we refactor the logic into a custom hook named useInfiniteScroll. 

import React, { useState, useEffect } from "react";

export default function useInfiniteScroll(loadDataFn) {
   const [bottom, setBottom] = useState(false);

   useEffect(() => {
      function handleScroll() {
         if (window.innerHeight + document.documentElement.scrollTop
            !== document.documentElement.offsetHeight) return;
         setBottom(true);
      }
      window.addEventListener("scroll", handleScroll);
      return () => window.removeEventListener("scroll", handleScroll);
   }, []);

   useEffect(() => {
      if (!bottom) return;
      loadDataFn();
   }, [bottom]);

   return [bottom, setBottom];
}

Step 3: Use the Custom Hook in a Component

Now we create TodoListUsingCustomHook and use useInfiniteScroll to handle loading more items. 

import React, { useState, useEffect } from "react";
import useInfiniteScroll from "../Hooks/useInfiniteScroll";

function TodoListUsingCustomHook() {
   const [data, setData] = useState([]);
   const [count, setCount] = useState(100);
   const [bottom, setBottom] = useInfiniteScroll(loadMoreData);

   const loadData = () => {
      let items = [];
      for (let i = 1; i <= count; i++) {
         items.push({ id: i, todo: "Todo Item " + i });
      }
      setData(items);
   };

   function loadMoreData() {
      let items = [];
      for (let i = 1; i <= count; i++) {
         items.push({ id: i, todo: "Todo Item " + i });
      }
      setData(items);
      setCount(count + 100);
      setBottom(false);
   }

   useEffect(() => {
      loadData(count);
   }, []);

   return (
      <div>
         <p>List of Todo list</p>
         <ul>
            {data.map(item => <li key={item.id}>{item.todo}</li>)}
         </ul>
      </div>
   );
}
export default TodoListUsingCustomHook;

Summary

Custom hooks allow developers to extract reusable logic from components. In this example, we successfully created useInfiniteScroll, which can now be reused in any React component requiring infinite scroll.
