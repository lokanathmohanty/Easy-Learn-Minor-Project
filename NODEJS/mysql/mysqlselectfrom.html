Node.js - MySQL Select From

Overview

Retrieving data from a MySQL database is one of the most common operations. The SELECT statement is used to fetch data, with optional clauses like WHERE, ORDER BY, and others. When executed in Node.js, it returns an array of objects, which can be JSON-ified for processing.

MySQL SELECT Syntax

SELECT field1, field2, ... FROM table_name
-- or --
SELECT * FROM table_name;

The * selects all columns.

Example: Basic SELECT in Node.js

var mysql = require('mysql');

var con = mysql.createConnection({
  host: "localhost",
  user: "root",
  password: "mypassword",
  database: "mydb"
});

var qry = `SELECT name, salary FROM employee;`;

con.connect(function (err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, function (err, results) {
    if (err) throw err;
    console.log(results);
  });
  con.end();
});

Sample output:

[ RowDataPacket { name: 'Ravi', salary: 25000 },
RowDataPacket { name: 'Anil', salary: 30000 },
RowDataPacket { name: 'Meena', salary: 27000 } ]

Returning JSON String

Use JSON.stringify() to convert result objects into strings:

con.query(qry, function (err, results) {
  if (err) throw err;
  console.log(JSON.stringify(results));
});

Result:

[{"name":"Ravi","salary":25000},{"name":"Anil","salary":30000},{"name":"Meena","salary":27000}]

Fetching All Rows and Columns

con.query("SELECT * FROM employee;", function (err, results) {
  if (err) throw err;
  results.forEach((row) => {
    console.log(`ID: ${row['id']}, NAME: ${row['name']}, AGE: ${row['age']}, SALARY: ${row['salary']}`);
  });
});

Sample output:

ID: 1, NAME: Ravi, AGE: 25, SALARY: 25000
ID: 2, NAME: Anil, AGE: 26, SALARY: 30000
ID: 3, NAME: Meena, AGE: 26, SALARY: 27000

Computed Columns with Aliases

Calculate derived data like taxes and name aliases:

con.query("SELECT name, salary, salary*0.05 as tax FROM employee;", function (err, results) {
  if (err) throw err;
  console.log(JSON.stringify(results));
});

Sample output:

[{"name":"Ravi","salary":25000,"tax":1250},
{"name":"Anil","salary":30000,"tax":1500},
{"name":"Meena","salary":27000,"tax":1350}]
