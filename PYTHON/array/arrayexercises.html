Python Array Exercises

Example 1: Find the Largest Number

import array as arr
a = arr.array('i', [10, 5, 15, 4, 6, 20, 9])
print(a)
largest = a[0]
for i in range(1, len(a)):
    if a[i] > largest:
        largest = a[i]
print("Largest number:", largest)
# Output:
# array('i', [10, 5, 15, 4, 6, 20, 9])
# Largest number: 20

Example 2: Store All Even Numbers

import array as arr
a = arr.array('i', [10, 5, 15, 4, 6, 20, 9])
print(a)
b = arr.array('i')
for i in range(len(a)):
    if a[i] % 2 == 0:
        b.append(a[i])
print("Even numbers:", b)
# Output:
# array('i', [10, 5, 15, 4, 6, 20, 9])
# Even numbers: array('i', [10, 4, 6, 20])

Example 3: Calculate Average

import array as arr
a = arr.array('i', [10, 5, 15, 4, 6, 20, 9])
print(a)
s = 0
for i in range(len(a)):
    s += a[i]
avg = s / len(a)
print("Average:", avg)
# Using sum() function
avg = sum(a) / len(a)
print("Average:", avg)
# Output:
# array('i', [10, 5, 15, 4, 6, 20, 9])
# Average: 9.857142857142858
# Average: 9.857142857142858
