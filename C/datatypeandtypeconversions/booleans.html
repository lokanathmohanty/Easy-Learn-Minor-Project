Booleans in C

The ANSI C standard originally did not have a built-in Boolean type. The Boolean type represents a value that can be one of two states: true or false. Though older C versions lacked this type, booleans can be implemented using enums or macros, and modern C (C99 and later) introduced the bool type in stdbool.h.

Implementing Boolean Using enum

Enumerations allow defining meaningful constants for true and false assigned to 1 and 0.

Example 1: Boolean with enum


#include <stdio.h>

int main() {
  enum bool {false, true};
  enum bool x = true;
  enum bool y = false;
  
  printf("%d\n", x);
  printf("%d\n", y);
}

Output:


1
0

Example 2: Typedef for Boolean


#include <stdio.h>

int main() {
  typedef enum {false, true} BOOL;
  
  BOOL x = true;
  BOOL y = false;
  
  printf("%d\n", x);
  printf("%d\n", y);
}

Output:


1
0

Boolean with Decisions and Loops


#include <stdio.h>

int main() {
  typedef enum {false, true} BOOL;
  int i = 0;
  
  while(true) {
    i++;
    printf("%d\n", i);
    if(i >= 5)
      break;
  }
  return 0;
}

Output:


1
2
3
4
5

Boolean with #define


#include <stdio.h>

#define FALSE 0
#define TRUE 1

int main() {
  printf("False: %d \nTrue: %d", FALSE, TRUE);
  return 0;
}

Output:


False: 0 
True: 1

Boolean with stdbool.h (C99 and later)


#include <stdio.h>
#include <stdbool.h>

int main() {
  bool a = true;
  bool b = false;
  
  printf("True: %d\n", a);
  printf("False: %d\n", b);
  
  return 0;
}

Output:


True: 1
False: 0

Example: Boolean in Logical Expressions


#include <stdio.h>
#include <stdbool.h>

int main() {
  bool x = 10 > 5;
  bool y = 40 > 50;
  
  if(x)
    printf("x is True\n");
  else
    printf("x is False\n");
  
  if(y)
    printf("Result: Pass\n");
  else
    printf("Result: Fail\n");
}

Output:


x is True
Result: Fail

Example: Boolean with While Loop


#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

int main(void) {
  bool loop = true;
  int i = 0;
  
  while(loop) {
    i++;
    printf("i: %d\n", i);
    if(i >= 5)
      loop = false;
  }
  
  printf("Loop stopped!\n");
  return EXIT_SUCCESS;
}

Output:


i: 1
i: 2
i: 3
i: 4
i: 5
Loop stopped!
