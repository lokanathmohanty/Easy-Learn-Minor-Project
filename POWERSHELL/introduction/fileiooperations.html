Example: Create Text File

Set-Content -Path "C:\example.txt" -Value "Hello, PowerShell!"

Example: Read Text File

Get-Content -Path "C:\example.txt"

Example: Create XML File

[xml]$xml = New-Object System.Xml.XmlDocument
$root = $xml.CreateElement("Root")
$xml.AppendChild($root) | Out-Null
$xml.Save("C:\example.xml")

Example: Read XML File

[xml]$xml = Get-Content -Path "C:\example.xml"

Example: Create CSV File

$data = @(
  @{Name="John"; Age=30}
  @{Name="Jane"; Age=28}
)
$data | Export-Csv -Path "C:\example.csv" -NoTypeInformation

Example: Read CSV File

Import-Csv -Path "C:\example.csv"

Example: Create HTML File

$data = Get-Process | Select-Object -First 5
$data | ConvertTo-Html | Out-File "C:\example.html"

Example: Read HTML File

Get-Content -Path "C:\example.html"

Example: Erase File Content

Clear-Content -Path "C:\example.txt"

Example: Append Text Data

Add-Content -Path "C:\example.txt" -Value "Appended text."
