HTML - CORS

Introduction

Cross-Origin Resource Sharing (CORS) is a mechanism that allows loading restricted resources from another domain in a web browser.

For example, when using the HTML5 video player, it may request camera permission. If the user allows it, the camera opens; otherwise, access is denied. Similarly, CORS ensures controlled access to external resources.

Making a CORS Request

Modern browsers like Chrome, Firefox, Opera, and Safari use the XMLHttpRequest2 object, while Internet Explorer uses a similar XDomainRequest object.

function createCORSRequest(method, url) {
   var xhr = new XMLHttpRequest();
   
   if ("withCredentials" in xhr) {
      xhr.open(method, url, true);
   } else if (typeof XDomainRequest != "undefined") {
      xhr = new XDomainRequest();
      xhr.open(method, url);
   } else {
      xhr = null;
   }
   return xhr;
}

var xhr = createCORSRequest('GET', url);
if (!xhr) {
   throw new Error('CORS not supported');
}

Event Handlers in CORS

  • onloadstart − Starts the request
  • onprogress − Loads and sends data
  • onabort − Aborts the request
  • onerror − Request has failed
  • onload − Request loaded successfully
  • ontimeout − Timeout occurred before completion
  • onloadend − Request completed (success or failure)

Example: Using onload and onerror

xhr.onload = function() {
   var responseText = xhr.responseText;
   console.log(responseText);
};

xhr.onerror = function() {
   console.log('There was an error!');
};

Example of CORS with Handler

The following example demonstrates a complete CORS request with onload and onerror handling:

// Create the XHR object.
function createCORSRequest(method, url) {
   var xhr = new XMLHttpRequest();
   if ("withCredentials" in xhr) {
      xhr.open(method, url, true);
   } else if (typeof XDomainRequest != "undefined") {
      xhr = new XDomainRequest();
      xhr.open(method, url);
   } else {
      xhr = null;
   }
   return xhr;
}

// Helper to parse the title tag.
function getTitle(text) {
   return text.match('<title>(.*)?</title>')[1];
}

// Make the actual CORS request.
function makeCorsRequest() {
   var url = 'http://www.tutorialspoint.com';
   var xhr = createCORSRequest('GET', url);
   if (!xhr) {
      alert('CORS not supported');
      return;
   }
   
   xhr.onload = function() {
      var text = xhr.responseText;
      var title = getTitle(text);
      alert('Response from CORS request to ' + url + ': ' + title);
   };

   xhr.onerror = function() {
      alert('Woops, there was an error making the request.');
   };

   xhr.send();
}
