Python - Singleton Class

A Singleton class in Python is a class that allows only a single instance to be created. This pattern is useful to optimize memory usage, for example, when creating a database connection.

Attempts to create multiple objects will return the same instance.

Singleton Using __init__ Method

class Singleton:
  __uniqueInstance = None

  @staticmethod
  def createInstance():
    if Singleton.__uniqueInstance is None:
      Singleton()
    return Singleton.__uniqueInstance

  def __init__(self):
    if Singleton.__uniqueInstance is not None:
      raise Exception("Object exist!")
    else:
      Singleton.__uniqueInstance = self

obj1 = Singleton.createInstance()
print(obj1)
obj2 = Singleton.createInstance()
print(obj2)

Output:

<__main__.Singleton object at 0x...>
<__main__.Singleton object at 0x...>

Singleton Using __new__ Method

class SingletonClass:
  _instance = None

  def __new__(cls):
    if cls._instance is None:
      print('Creating the object')
      cls._instance = super(SingletonClass, cls).__new__(cls)
    return cls._instance

obj1 = SingletonClass()
print(obj1)

obj2 = SingletonClass()
print(obj2)

Output:

Creating the object
<__main__.SingletonClass object at 0x...>
<__main__.SingletonClass object at 0x...>
