JavaScript and Cookies

What are Cookies?

In JavaScript, cookies are pieces of data stored in the user's web browser. They are stored as key-value pairs and can be manipulated using the cookie property of the document object. You can set, read, update, and delete cookies from JavaScript.

Why are Cookies Needed?

HTTP is a stateless protocol, meaning it does not retain user session information across requests. Cookies help websites remember user sessions, preferences, logins, and activities across multiple pages.

Examples of use cases include:

  • Remembering user login state
  • Tracking purchases or shopping cart data
  • Storing visitor preferences
  • Caching for better performance

How It Works

When a server sends data to a browser, it is stored as a cookie (plain text). On subsequent requests, the browser automatically sends the cookie back to the server. Cookies typically contain these fields:

  • Expires – Date when the cookie will expire.
  • Domain – The domain where the cookie is valid.
  • Path – The path of the page or directory for which the cookie is valid.
  • Secure – Restricts cookies to secure connections if set.
  • Name=Value – Data stored in key-value pairs.

Setting / Storing Cookies

The simplest way to create a cookie is to assign a string value to document.cookie:

document.cookie = "key1=value1; key2=value2; expires=date";

Example:

<script>
function WriteCookie() {
  if (document.myform.customer.value == "") {
    alert("Enter some value!");
    return;
  }
  let cookievalue = escape(document.myform.customer.value) + ";";
  document.cookie = "name=" + cookievalue;
  document.write("Setting Cookies : name=" + cookievalue);
}
</script>

Reading Cookies

You can read cookies using document.cookie, which returns all cookies as a single string of key-value pairs separated by semicolons.

<script>
function ReadCookie() {
  let allcookies = document.cookie;
  document.write("All Cookies : " + allcookies);

  let cookiearray = allcookies.split(';');
  for (let i = 0; i < cookiearray.length; i++) {
    let name = cookiearray[i].split('=')[0];
    let value = cookiearray[i].split('=')[1];
    document.write("Key is : " + name + " and Value is : " + value);
  }
}
</script>

Setting Cookies Expiry Date

You can extend cookie lifetime by adding the expires attribute:

<script>
function WriteCookie() {
  let now = new Date();
  now.setMonth(now.getMonth() + 1);

  let cookievalue = escape(document.myform.customer.value) + ";";
  document.cookie = "name=" + cookievalue;
  document.cookie = "expires=" + now.toUTCString() + ";";
  document.write("Setting Cookies : name=" + cookievalue);
}
</script>

Deleting a Cookie

To delete a cookie, set its expiry date to a past date:

<script>
function WriteCookie() {
  let now = new Date();
  now.setMonth(now.getMonth() - 1);

  let cookievalue = escape(document.myform.customer.value) + ";";
  document.cookie = "name=" + cookievalue;
  document.cookie = "expires=" + now.toUTCString() + ";";
  document.write("Deleting Cookie : name=" + cookievalue);
}
</script>

Updating Cookies

Updating a cookie is done by setting the same key with a new value:

document.cookie = "key1=newValue";

Example:

<script>
let output = document.getElementById("output");

function setCookies() {
  document.cookie = "cartItem=watch";
  document.cookie = "price=10000";
}

function updateCookies() {
  document.cookie = "cartItem=bag";
  document.cookie = "price=5000";
}

function getCookies() {
  let allCookies = document.cookie.split("; ");
  output.innerHTML = "The cookie data are : <br>";
  for (const cookie of allCookies) {
    const [key, value] = cookie.split("=");
    if (key == "cartItem" || key == "price") {
      output.innerHTML += key + " : " + decodeURIComponent(value) + "<br>";
    }
  }
}
</script>
