R - Vectors

Vectors are the most basic R data objects, available in six atomic types: logical, integer, double, complex, character, and raw.

Vector Creation

Single Element Vector

# Character
print("abc")
# Double
print(12.5)
# Integer
print(63L)
# Logical
print(TRUE)
# Complex
print(2+3i)
# Raw
print(charToRaw('hello'))

Output: 

[1] "abc"
[1] 12.5
[1] 63
[1] TRUE
[1] 2+3i
[1] 68 65 6c 6c 6f

Multiple Elements Vector

Using colon operator: 

v <- 5:13
print(v)
v <- 6.6:12.6
print(v)
v <- 3.8:11.4
print(v)

Output: 

[1]  5  6  7  8  9 10 11 12 13
[1]  6.6  7.6  8.6  9.6 10.6 11.6 12.6
[1]  3.8  4.8  5.8  6.8  7.8  8.8  9.8 10.8

Using sequence operator: 

print(seq(5, 9, by = 0.4))

Output: 

[1] 5.0 5.4 5.8 6.2 6.6 7.0 7.4 7.8 8.2 8.6 9.0

Using c() function: 

s <- c('apple','red',5,TRUE)
print(s)

Output: 

[1] "apple" "red"   "5"     "TRUE"

Accessing Vector Elements

t <- c("Sun","Mon","Tue","Wed","Thurs","Fri","Sat")
u <- t[c(2,3,6)]
print(u)
v <- t[c(TRUE,FALSE,FALSE,FALSE,FALSE,TRUE,FALSE)]
print(v)
x <- t[c(-2,-5)]
print(x)
y <- t[c(0,0,0,0,0,0,1)]
print(y)

Output: 

[1] "Mon" "Tue" "Fri"
[1] "Sun" "Fri"
[1] "Sun" "Tue" "Wed" "Fri" "Sat"
[1] "Sun"

Vector Manipulation

Vector Arithmetic

v1 <- c(3,8,4,5,0,11)
v2 <- c(4,11,0,8,1,2)
add.result <- v1 + v2
print(add.result)
sub.result <- v1 - v2
print(sub.result)
multi.result <- v1 * v2
print(multi.result)
divi.result <- v1 / v2
print(divi.result)

Output: 

[1]  7 19  4 13  1 13
[1] -1 -3  4 -3 -1  9
[1] 12 88  0 40  0 22
[1] 0.7500000 0.7272727       Inf 0.6250000 0.0000000 5.5000000

Vector Element Recycling

v1 <- c(3,8,4,5,0,11)
v2 <- c(4,11)
add.result <- v1 + v2
print(add.result)
sub.result <- v1 - v2
print(sub.result)

Output: 

[1]  7 19  8 16  4 22
[1] -1 -3  0 -6 -4  0

Vector Element Sorting

v <- c(3,8,4,5,0,11, -9, 304)
sort.result <- sort(v)
print(sort.result)
revsort.result <- sort(v, decreasing = TRUE)
print(revsort.result)
v <- c("Red","Blue","yellow","violet")
sort.result <- sort(v)
print(sort.result)
revsort.result <- sort(v, decreasing = TRUE)
print(revsort.result)

Output: 

[1]  -9   0   3   4   5   8  11 304
[1] 304  11   8   5   4   3   0  -9
[1] "Blue"   "Red"    "violet" "yellow"
[1] "yellow" "violet" "Red"    "Blue"
