Node.js - MySQL Update

Overview

The UPDATE statement in MySQL allows modifying one or more records in a table. In Node.js, this is achieved by passing the UPDATE query string to the connection's query method.

Basic UPDATE Syntax

UPDATE table_name
SET column1 = expr1, column2 = expr2, ...
WHERE condition;

The WHERE clause restricts the rows to update.

Simple UPDATE Example

var mysql = require('mysql');
var con = mysql.createConnection({
  host: "localhost",
  user: "root",
  password: "mypassword",
  database: "mydb"
});

var qry = "UPDATE employee SET salary = salary + 500;";

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, function(err) {
    if (err) throw err;
    console.log("Records updated successfully");
  });
  con.end();
});

Check updated data:

SELECT name, salary FROM employee;

UPDATE with Prepared Statement

var qry = "UPDATE employee SET salary = ? WHERE name = ?";
var params = [40000, "Anil"];

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, params, function(err) {
    if (err) throw err;
    console.log("Record updated successfully");

    // Verify updated record
    var verifyQry = "SELECT name, salary FROM employee WHERE name = ?";
    con.query(verifyQry, [params[1]], function(err, results) {
      if (err) throw err;
      console.log(results);
    });
  });
  con.end();
});

UPDATE with JOIN

Use JOIN in UPDATE to modify fields in one table based on related data in another table. 

var qry = `
UPDATE employees
INNER JOIN merits ON employees.performance = merits.performance
SET salary = salary + salary * percentage;
`;

con.connect(function(err) {
  if (err) throw err;
  console.log("Connected!");
  con.query(qry, function(err) {
    if (err) throw err;

    // Check updated employees
    var checkQry = "SELECT * FROM employees";
    con.query(checkQry, function(err, results) {
      if (err) throw err;
      console.log(results);
    });
  });
  con.end();
});
