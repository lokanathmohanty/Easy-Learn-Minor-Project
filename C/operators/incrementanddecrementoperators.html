Increment and Decrement Operators in C

The increment operator (++) increases the value of a variable by 1, while the decrement operator (--) decreases the value by 1. These unary operators can be used as a prefix or postfix.

Basic Example

The following example shows the use of increment and decrement operators with different variations:

#include <stdio.h>

int main() {
  int a = 5, b = 5, c = 5, d = 5;
  
  a++; // postfix increment
  ++b; // prefix increment
  c--; // postfix decrement
  --d; // prefix decrement

  printf("a = %d\n", a);
  printf("b = %d\n", b);
  printf("c = %d\n", c);
  printf("d = %d\n", d);

  return 0;
}

Output:

a = 6
b = 6
c = 4
d = 4

Prefix vs Postfix Increment

The prefix increment operator increases the value before using it, while the postfix increment increases the value after using it.

#include <stdio.h>

int main() {
  int x = 10;

  int y = 10 + ++x;  // prefix increment
  printf("x = %d, y = %d\n", x, y);

  y = 10 + x++;      // postfix increment
  printf("x = %d, y = %d\n", x, y);

  return 0;
}

Output:

x = 11, y = 21
x = 12, y = 21

Prefix vs Postfix Decrement

The prefix decrement operator decreases the value before using it, while the postfix decrement decreases the value after using it.

#include <stdio.h>

int main() {
  int x = 10;

  int y = 10 + --x;  // prefix decrement
  printf("x = %d, y = %d\n", x, y);

  y = 10 + x--;      // postfix decrement
  printf("x = %d, y = %d\n", x, y);

  return 0;
}

Output:

x = 9, y = 19
x = 8, y = 19

More Examples

The following example demonstrates prefix and postfix increments and decrements with char and int types:

#include <stdio.h>

int main() {
  char a = 'a', b = 'M';
  int x = 5, y = 23;

  printf("a: %c b: %c\n", a, b);
  a++;
  printf("postfix increment a: %c\n", a);

  ++b;
  printf("prefix increment b: %c\n", b);

  printf("x: %d y: %d\n", x, y);
  x--;
  printf("postfix decrement x: %d\n", x);

  --y;
  printf("prefix decrement y: %d\n", y);

  return 0;
}

Output:

a: a b: M
postfix increment a: b
prefix increment b: N
x: 5 y: 23
postfix decrement x: 4
prefix decrement y: 22

Operator Precedence

Postfix increment and decrement have higher precedence than prefix versions. Assignment operator has lower precedence than prefix increment.

#include <stdio.h>

int main() {
  int x = 5, z;

  z = x++;
  printf("x: %d z: %d\n", x, z);

  int y = 5;
  z = ++y;
  printf("y: %d z: %d\n", y, z);

  return 0;
}

Output:

x: 6 z: 5
y: 6 z: 6

Multiple Operators in Expression

#include <stdio.h>

int main() {
  int x = 5, y = 5, z;

  z = x++ + ++y;
  printf("x: %d y: %d z: %d\n", x, y, z);

  return 0;
}

Output:

x: 6 y: 6 z: 11

Using Increment in Loop

#include <stdio.h>

int main() {
  for (int x = 1; x <= 5; x++) {
    printf("x: %d\n", x);
  }
  return 0;
}

Output:

x: 1
x: 2
x: 3
x: 4
x: 5
