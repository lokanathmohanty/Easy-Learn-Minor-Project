Python - Inter-Thread Communication

In Python, threads can communicate and synchronize using shared data and synchronization primitives like Event and Condition objects from the threading module.

Using Event Object

An Event object manages a flag, allowing threads to wait for and notify about changes. The flag is initially false and can be set or cleared by threads.

Key methods: set(), clear(), wait(), is_set().

from threading import Event, Thread
import time

terminate = False

def signal_state():
    global terminate
    while not terminate:
        time.sleep(0.5)
        print('Traffic Police: GREEN light')
        event.set()
        time.sleep(1)
        print('Traffic Police: RED light')
        event.clear()

def traffic_flow():
    global terminate
    num = 0
    while num < 10 and not terminate:
        print('Waiting for GREEN light')
        event.wait()
        print('GREEN light - vehicles moving')
        while event.is_set() and not terminate:
            num += 1
            print('Vehicle No:', num, 'Crossing')
            time.sleep(1)
        print('RED light - vehicles stopped')

event = Event()
t1 = Thread(target=signal_state)
t2 = Thread(target=traffic_flow)

t1.start()
t2.start()

time.sleep(5)
terminate = True

t1.join()
t2.join()
print('Exiting')

Output:

Waiting for GREEN light
Traffic Police: GREEN light
GREEN light - vehicles moving
Vehicle No: 1 Crossing
Traffic Police: RED light
RED light - vehicles stopped
Waiting for GREEN light
...

Using Condition Object

Condition object allows threads to wait for a condition and notify others upon changes. It is used with an underlying lock.

from threading import Condition, Thread
import time

c = Condition()

def thread_a():
    print('Thread A started')
    with c:
        print('Thread A waiting')
        c.wait()
        print('Thread A notified')

def thread_b():
    print('Thread B started')
    with c:
        time.sleep(2)
        print('Thread B notifying')
        c.notify()

Thread(target=thread_a).start()
Thread(target=thread_b).start()

Output:

Thread A started
Thread A waiting
Thread B started
Thread B notifying
Thread A notified

Producer-Consumer with Condition

Example where one thread produces data and signals the consumer thread to consume it using Condition.

from threading import Condition, Thread
import time
import random

numbers = []

def producer(c):
    for _ in range(5):
        with c:
            num = random.randint(1, 10)
            numbers.append(num)
            print('Produced:', num)
            c.notify()
        time.sleep(0.3)

def consumer(c):
    for _ in range(5):
        with c:
            while not numbers:
                c.wait()
            num = numbers.pop()
            print('Consumed:', num)
        time.sleep(0.3)

cond = Condition()
t1 = Thread(target=producer, args=(cond,))
t2 = Thread(target=consumer, args=(cond,))

t1.start()
t2.start()
t1.join()
t2.join()

print('Done')

Output:

Produced: 3
Consumed: 3
Produced: 7
Consumed: 7
