Node.js - MongoDB Delete

Overview

MongoDB provides two methods to delete documents in collections via the Node.js driver: deleteOne() to remove a single matching document, and deleteMany() to remove all matching documents.

deleteOne() Example

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);

  try {
    await client.connect();
    await deleteDoc(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function deleteDoc(client, dbname, colname) {
  const filter = { Name: "TV" };
  const result = await client.db(dbname).collection(colname).deleteOne(filter);
  console.log("Document Deleted");
}

main().catch(console.error);

Use MongoDB shell or Compass to verify deletion.

deleteMany() Example

async function deleteDocs(client, dbname, colname) {
  const filter = { price: { $gt: 10000 } };
  const result = await client.db(dbname).collection(colname).deleteMany(filter);
  console.log("Documents Deleted");
}

Dropping a Collection

Remove an entire collection with drop() or dropCollection():

async function dropCollection(client, dbname, colname) {
  const result = await client.db(dbname).collection(colname).drop();
  console.log("Collection dropped");
}

Or alternatively:

async function dropCollectionAlt(client, dbname, colname) {
  const result = await client.db(dbname).dropCollection(colname);
  console.log("Collection dropped");
}

Verify collection removal using MongoDB shell.
