Node.js - MongoDB Find

Overview

In MongoDB, the find() and findOne() methods are used to query documents, similar to SQL's SELECT statement. These methods return matching documents from a collection based on a filter.

Finding All Documents

Use find({}) to retrieve all documents in a collection.

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);
  try {
    await client.connect();
    await listAll(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function listAll(client, dbName, colName) {
  const result = await client.db(dbName).collection(colName).find({}).toArray();
  console.log(JSON.stringify(result));
}

main().catch(console.error);

Sample output:

[{"_id":"65809214693bd4622484dce3","ProductID":1,"Name":"Laptop","Price":25000}, ...]

Traversing the Resultset

Use a loop, such as forEach, to process documents:

result.forEach(row => {
  console.log(row['Name'], row['Price']);
});

Sample output:

Desktop 20000
Laptop 25000
TV 40000
Router 2000
Scanner 5000
Printer 9000

findOne() Method

Returns the first document matching the filter:

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);
  try {
    await client.connect();
    await findOneDoc(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function findOneDoc(client, dbName, colName) {
  const result = await client.db(dbName).collection(colName).findOne({ Name: "TV" });
  console.log(JSON.stringify(result));
}

main().catch(console.error);

Sample output:

{"_id":"6580964f20f979d2e9a72ae8","ProductID":2,"Name":"TV","Price":40000}

Without filter, findOne({}) returns the first document in the collection.

const { MongoClient } = require('mongodb');

async function main() {
  const uri = "mongodb://localhost:27017/";
  const client = new MongoClient(uri);
  try {
    await client.connect();
    await findOneDoc(client, "mydb", "products");
  } finally {
    await client.close();
  }
}

async function findOneDoc(client, dbName, colName) {
  const result = await client.db(dbName).collection(colName).findOne({});
  console.log(JSON.stringify(result));
}

main().catch(console.error);

Sample output:

{"_id":"65809214693bd4622484dce3","ProductID":1,"Name":"Laptop","Price":25000}
