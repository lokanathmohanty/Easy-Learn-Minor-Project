R - Lists

Lists in R are objects that can contain elements of different types, such as numbers, strings, vectors, or even other lists, matrices, and functions. Lists are created using the list() function.

Creating a List

list_data <- list("Red", "Green", c(21,32,11), TRUE, 51.23, 119.1)
print(list_data)

Output:

[[1]]
[1] "Red"
[[2]]
[1] "Green"
[[3]]
[1] 21 32 11
[[4]]
[1] TRUE
[[5]]
[1] 51.23
[[6]]
[1] 119.1

Naming List Elements

list_data <- list(
  c("Jan","Feb","Mar"),
  matrix(c(3,9,5,1,-2,8), nrow = 2),
  list("green",12.3)
)
names(list_data) <- c("1st Quarter", "A_Matrix", "A Inner list")
print(list_data)

Output:

$`1st_Quarter`
[1] "Jan" "Feb" "Mar"
$A_Matrix
     [,1] [,2] [,3]
[1,]    3    5   -2
[2,]    9    1    8
$A_Inner_list
$A_Inner_list[[1]]
[1] "green"
$A_Inner_list[[2]]
[1] 12.3

Accessing List Elements

# Index-based access:
print(list_data[1])
print(list_data[3])
# Name-based access:
print(list_data$A_Matrix)

Output:

$`1st_Quarter`
[1] "Jan" "Feb" "Mar"
$A_Inner_list
$A_Inner_list[[1]]
[1] "green"
$A_Inner_list[[2]]
[1] 12.3
     [,1] [,2] [,3]
[1,]    3    5   -2
[2,]    9    1    8

Manipulating List Elements

# Add element at the end
list_data[4] <- "New element"
print(list_data[4])
# Remove last element
list_data[4] <- NULL
print(list_data[4])
# Update 3rd element
list_data[3] <- "updated element"
print(list_data[3])

Output:

[[1]]
[1] "New element"
$<NA>
NULL
$`A Inner list`
[1] "updated element"

Merging Lists

list1 <- list(1,2,3)
list2 <- list("Sun","Mon","Tue")
merged.list <- c(list1, list2)
print(merged.list)

Output:

[[1]]
[1] 1
[[2]]
[1] 2
[[3]]
[1] 3
[[4]]
[1] "Sun"
[[5]]
[1] "Mon"
[[6]]
[1] "Tue"

Converting List to Vector

list1 <- list(1:5)
print(list1)
list2 <- list(10:14)
print(list2)
v1 <- unlist(list1)
v2 <- unlist(list2)
print(v1)
print(v2)
result <- v1 + v2
print(result)

Output:

[[1]]
[1] 1 2 3 4 5
[[1]]
[1] 10 11 12 13 14
[1] 1 2 3 4 5
[1] 10 11 12 13 14
[1] 11 13 15 17 19
