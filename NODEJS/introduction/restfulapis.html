Node.js - RESTful API with Express.js

Overview

ExpressJS is ideal for building REST APIs exposing resources via URIs to perform read/write operations over HTTP. REST architecture defines resource-based endpoints with stateless interactions using HTTP methods.

REST Architecture

  • Uniform interface
  • Stateless
  • Client-server separation
  • Cacheable
  • Layered system
  • Code on demand

Popular representations include JSON (most commonly) and XML.

Common HTTP Methods in REST

MethodOperationExample
POSTCreate new resourcePOST /users
GETRead existing resourceGET /users or /users/123
PUTUpdate resourcePUT /users/123
DELETEDelete resourceDELETE /users/123

Example User RESTful API

var express = require('express');
var app = express();
var fs = require('fs');
var bodyParser = require('body-parser');

app.use(bodyParser.json());
app.use(bodyParser.urlencoded({ extended: true }));

// List all users
app.get('/', function (req, res) {
  fs.readFile(__dirname + '/users.json', 'utf8', function (err, data) {
    res.end(data);
  });
});

// Show user detail
app.get('/:id', function (req, res) {
  fs.readFile(__dirname + '/users.json', 'utf8', function (err, data) {
    var users = JSON.parse(data);
    var user = users['user' + req.params.id];
    res.end(JSON.stringify(user));
  });
});

// Add new user
app.post('/', function (req, res) {
  fs.readFile(__dirname + '/users.json', 'utf8', function (err, data) {
    var users = JSON.parse(data);
    var user = req.body.user4;
    users['user' + user.id] = user;
    res.end(JSON.stringify(users));
  });
});

// Delete user
app.delete('/:id', function (req, res) {
  fs.readFile(__dirname + '/users.json', 'utf8', function (err, data) {
    var users = JSON.parse(data);
    delete users['user' + req.params.id];
    res.end(JSON.stringify(users));
  });
});

// Update user
app.put('/:id', function (req, res) {
  fs.readFile(__dirname + '/users.json', 'utf8', function (err, data) {
    var users = JSON.parse(data);
    users['user' + req.params.id] = req.body;
    res.end(JSON.stringify(users));
  });
});

var server = app.listen(5000, function () {
  console.log('Express App running at http://127.0.0.1:5000/');
});

Testing

Use REST clients like Postman or Insomnia to test the endpoints by sending GET, POST, DELETE, PUT requests to http://localhost:5000/ and checking JSON responses.
