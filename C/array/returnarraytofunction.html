Return Array from Function in C

Functions in C can return values to the caller. Although C functions typically return a single value, you can return an array using various techniques:

  • Passing an array and returning its pointer
  • Returning a static array
  • Returning dynamically allocated array with malloc()
  • Embedding the array in a struct

Pass Array by Reference

An uninitialized array is passed to a function, and its pointer is returned. The array is modified inside the function.

#include <stdio.h>
#include <math.h>

int arrfunction(int, float *);

int main() {
   int x = 100;
   float arr[3];
   arrfunction(x, arr);
   printf("Square of %d: %f\n", x, arr[0]);
   printf("cube of %d: %f\n", x, arr[1]);
   printf("Square root of %d: %f\n", x, arr[2]);
   return 0;
}

int arrfunction(int x, float *arr) {
   arr[0] = pow(x, 2);
   arr[1] = pow(x, 3);
   arr[2] = pow(x, 0.5);
}

Output:

Square of 100: 10000.000000
cube of 100: 1000000.000000
Square root of 100: 10.000000

Return Static Array

A static array inside the function is used to store the values, and its pointer is returned. This avoids returning a local variable pointer.

#include <stdio.h>
#include <math.h>

float * arrfunction(int);

int main() {
   int x = 100;
   float *arr = arrfunction(x);
   printf("Square of %d: %f\n", x, arr[0]);
   printf("cube of %d: %f\n", x, arr[1]);
   printf("Square root of %d: %f\n", x, arr[2]);
   return 0;
}

float *arrfunction(int x) {
   static float arr[3];
   arr[0] = pow(x, 2);
   arr[1] = pow(x, 3);
   arr[2] = pow(x, 0.5);
   return arr;
}

Output:

Square of 100: 10000.000000
cube of 100: 1000000.000000
Square root of 100: 10.000000

Return Array of Random Numbers

This example shows how to return a static array filled with random integers.

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int *getRandom() {
   static int r[10];
   int i;
   srand((unsigned)time(NULL));
   for (i = 0; i < 10; ++i) {
      r[i] = rand();
      printf("r[%d] = %d\n", i, r[i]);
   }
   return r;
}

int main() {
   int *p = getRandom();
   for (int i = 0; i < 10; i++) {
      printf("*(p + %d) : %d\n", i, *(p + i));
   }
   return 0;
}

Sample Output:

r[0] = 2110147662
r[1] = 1427553496
...
*(p + 0) : 2110147662
*(p + 1) : 1427553496
...

Return Array Using malloc()

malloc() dynamically allocates memory during runtime. Always remember to free it if no longer needed.

#include <stdio.h>
#include <math.h>
#include <stdlib.h>

float * arrfunction(int);

int main() {
   int x = 16;
   float *arr = arrfunction(x);
   printf("Square of %d: %f\n", x, arr[0]);
   printf("cube of %d: %f\n", x, arr[1]);
   printf("Square root of %d: %f\n", x, arr[2]);
   free(arr);  // Optional cleanup
   return 0;
}

float *arrfunction(int x) {
   float *arr = (float *)malloc(3 * sizeof(float));
   arr[0] = pow(x, 2);
   arr[1] = pow(x, 3);
   arr[2] = pow(x, 0.5);
   return arr;
}

Output:

Square of 16: 256.000000
cube of 16: 4096.000000
Square root of 16: 4.000000

Return Array Inside a Struct

Embedding arrays inside struct is a clean and type-safe way to return multiple values.

#include <stdio.h>
#include <math.h>

struct mystruct {
   float arr[3];
};

struct mystruct myfunction(int x);

int main() {
   int x = 9;
   struct mystruct s = myfunction(x);
   printf("Square of %d: %f\n", x, s.arr[0]);
   printf("cube of %d: %f\n", x, s.arr[1]);
   printf("Square root of %d: %f\n", x, s.arr[2]);
   return 0;
}

struct mystruct myfunction(int x) {
   struct mystruct s1;
   s1.arr[0] = pow(x, 2);
   s1.arr[1] = pow(x, 3);
   s1.arr[2] = pow(x, 0.5);
   return s1;
}

Output:

Square of 9: 81.000000
cube of 9: 729.000000
Square root of 9: 3.000000

Return String from Function

Strings in C are character arrays. Use dynamic memory or static storage to safely return strings from functions.

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char * hellomsg(char *);

int main() {
   char *name = "TutorialsPoint";
   char *arr = hellomsg(name);
   printf("%s\n", arr);
   free(arr); // free dynamically allocated memory
   return 0;
}

char * hellomsg(char *x) {
   char *arr = (char *)malloc(50 * sizeof(char));
   strcpy(arr, "Hello ");
   strcat(arr, x);
   return arr;
}

Output:

Hello TutorialsPoint
