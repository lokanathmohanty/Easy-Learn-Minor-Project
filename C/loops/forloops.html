For Loop in C

The for loop is an entry-controlled counted loop, often the preferred loop type in C. It combines initialization, test condition, and increment in one statement.

Syntax

for (init; condition; increment){
   statement(s);
}

Control Flow

The initialization step runs once, then the condition is checked. If true, the loop body executes, followed by the increment step. This repeats until the condition is false.

Example: Basic for loop

#include <stdio.h>

int main(){
   int a;
   for(a = 1; a <= 5; a++){
      printf("a: %d\n", a);
   }
   return 0;
}

Output:

a: 1
a: 2
a: 3
a: 4
a: 5

Example: Initialization Outside Loop

#include <stdio.h>

int main(){
   int a = 1;
   for(; a <= 5; a++){
      printf("a: %d\n", a);
   }
   return 0;
}

Output is the same as above.

Example: Increment Inside Loop

#include <stdio.h>

int main(){
   int a;
   for(a = 1; a <= 5; ){
      printf("a: %d\n", a);
      a++;
   }
   return 0;
}

Output is the same as above.

Example: Omitted Condition with break

#include <stdio.h>

int main(){
   int a;
   for(a = 1; ; a++){
      printf("a: %d\n", a);
      if(a == 5)
         break;
   }
   return 0;
}

Output:

a: 1
a: 2
a: 3
a: 4
a: 5

Example: Multiple Counters

#include <stdio.h>

int main(){
   int a, b;
   for(a = 1, b = 1; a <= 5; a++, b++){
      printf("a: %d b: %d a*b: %d\n", a, b, a*b);
   }
   return 0;
}

Output:

a: 1 b: 1 a*b: 1
a: 2 b: 2 a*b: 4
a: 3 b: 3 a*b: 9
a: 4 b: 4 a*b: 16
a: 5 b: 5 a*b: 25

Example: Decrementing for Loop

#include <stdio.h>

int main(){
   int a;
   for(a = 5; a >= 1; a--){
      printf("a: %d\n", a);
   }
   return 0;
}

Output:

a: 5
a: 4
a: 3
a: 2
a: 1

Example: Traversing an Array

#include <stdio.h>

int main(){
   int i;
   int arr[] = {10, 20, 30, 40, 50};
   for(i = 0; i < 5; i++){
      printf("a[%d]: %d\n", i, arr[i]);
   }
   return 0;
}

Output:

a[0]: 10
a[1]: 20
a[2]: 30
a[3]: 40
a[4]: 50

Example: Sum and Average

#include <stdio.h>

int main(){
   int i;
   int arr[] = {10, 20, 30, 40, 50};
   int sum = 0;
   float avg;

   for(i=0; i < 5; i++){
      sum += arr[i];
   }
   avg = (float)sum / 5;
   printf("Average = %f", avg);
   return 0;
}

Output:
Average = 30.000000

Example: Factorial

#include <stdio.h>

int main(){
   int i, x = 5;
   int fact = 1;
   for(i=1; i <= x; i++){
      fact *= i;
   }
   printf("%d! = %d", x, fact);
   return 0;
}

Output:
5! = 120
